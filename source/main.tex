% 図は手順との対応を保つため[H]固定。かわりに手順説明はキャプションへ統合し、
% 図の幅と間隔を抑えて1ページに複数枚収まるようにして余白を防ぐ。
\newcommand{\installfigure}[4][0.60\linewidth]{%
  \begin{figure}[H]
    \centering
    \includegraphics[width=#1]{#2}
    \caption{#3}
    \label{#4}
  \end{figure}
}
% 図を左右2枚組で並べる版。縦積みを抑えてページ数を減らすために使う。
% #1: 各画像の幅（省略時はminipage幅いっぱい）、続けて（画像,キャプション,ラベル）を2組。
\newcommand{\installfigrow}[7][\linewidth]{%
  \begin{figure}[H]
    \centering
    \begin{minipage}[t]{0.48\linewidth}
      \centering
      \includegraphics[width=#1]{#2}
      \caption{#3}\label{#4}
    \end{minipage}\hfill
    \begin{minipage}[t]{0.48\linewidth}
      \centering
      \includegraphics[width=#1]{#5}
      \caption{#6}\label{#7}
    \end{minipage}
  \end{figure}
}
